  string(FIND "${${variable}}" "${needle}" found)
  if(found LESS 0)

  endif()

  elseif(SHADER STREQUAL "glass_ps")

    set(PROFILE ps_5_0)
    set(ENTRY PS)
  elseif(SHADER STREQUAL "glass_basic_ps")
    set(SOURCE glass.hlsl)
    set(PROFILE ps_5_0)
    set(ENTRY PSBasic)
  elseif(SHADER STREQUAL "backdrop_edge_ps")
    set(SOURCE backdrop_edge.hlsl)
    set(PROFILE ps_5_0)
    set(ENTRY PS)
  else()
    set(SOURCE compare.hlsl)
    set(PROFILE cs_5_0)
    set(ENTRY CS)
  endif()
  set(OUT "${CMAKE_CURRENT_BINARY_DIR}/${SHADER}.h")
  set(INPUT "${SHADER_DIR}/${SOURCE}")
  if(SOURCE STREQUAL "glass.hlsl")
    set(INPUT "${GLASS_SHADER_SOURCE}")
  endif()
  add_custom_command(OUTPUT "${OUT}" COMMAND "${FXC}" /nologo /O3 /WX /T "${PROFILE}" /E "${ENTRY}" /Vn "g_${SHADER}" /Fh "${OUT}" "${INPUT}" DEPENDS "${INPUT}" VERBATIM)
  list(APPEND SHADER_HEADERS "${OUT}")
endforeach()
